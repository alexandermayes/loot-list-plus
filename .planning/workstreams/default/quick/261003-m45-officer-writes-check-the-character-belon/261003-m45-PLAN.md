## Production count query (read-only, aggregate counts only; the user runs it in the Supabase SQL editor, never the executor; do not break the counts down by guild anywhere shared)

```sql
-- 1. Before merging: existing rows whose character has no active membership, or no membership row,
--    in the row's guild. The migration does not check or change them, they stay editable,
--    and a later change of their character or guild would be refused.
select 'raid_team_members' as table_name, count(*) as rows_with_character,
       count(*) filter (where not exists (select 1 from public.character_guild_memberships m
                         where m.character_id = t.character_id and m.guild_id = t.guild_id and m.is_active)) as no_active_membership,
       count(*) filter (where not exists (select 1 from public.character_guild_memberships m
                         where m.character_id = t.character_id and m.guild_id = t.guild_id)) as no_membership_row
  from public.raid_team_members t
union all
select 'donation_records', count(*),
       count(*) filter (where not exists (select 1 from public.character_guild_memberships m
                         where m.character_id = t.character_id and m.guild_id = t.guild_id and m.is_active)),
       count(*) filter (where not exists (select 1 from public.character_guild_memberships m
                         where m.character_id = t.character_id and m.guild_id = t.guild_id))
  from public.donation_records t
 where t.character_id is not null
union all
select 'attendance_records', count(*),
       count(*) filter (where not exists (select 1 from public.character_guild_memberships m
                         where m.character_id = t.character_id and m.guild_id = re.guild_id and m.is_active)),
       count(*) filter (where not exists (select 1 from public.character_guild_memberships m
                         where m.character_id = t.character_id and m.guild_id = re.guild_id))
  from public.attendance_records t
  join public.raid_events re on re.id = t.raid_event_id
 where t.character_id is not null;

-- 2. After the deploy: the three triggers (expected: attendance_records, donation_records, raid_team_members)
select c.relname as table_name, t.tgname as trigger_name
  from pg_trigger t
  join pg_class c on c.oid = t.tgrelid
  join pg_namespace n on n.oid = c.relnamespace
 where n.nspname = 'public' and not t.tgisinternal
   and t.tgname in ('enforce_guild_character_refs', 'enforce_attendance_character_refs')
 order by 1;

-- 3. After the deploy: EXECUTE on the two functions (expected anon false, authenticated false, service_role true)
select p.oid::regprocedure::text as function,
       has_function_privilege('anon', p.oid, 'EXECUTE') as anon,
       has_function_privilege('authenticated', p.oid, 'EXECUTE') as authenticated,
       has_function_privilege('service_role', p.oid, 'EXECUTE') as service_role
  from pg_proc p
  join pg_namespace n on n.oid = p.pronamespace
 where n.nspname = 'public'
   and p.proname in ('enforce_guild_character_refs', 'enforce_attendance_character_refs')
 order by 1;
```
